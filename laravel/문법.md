@csrf
	공격자가 사용자 권한을 이용해 의도치 않은 서버 공격을 하는 것을 방지하기 위해 숨겨진 토큰을 사용해서 요청 위조 방지를 한다. form태그 아래 넣지 않으면 http 419 error가 발생한다.

form의 내용을 전달받은 백단
```php
    public function store(Request $request)
    {
        //
        $request->validate([
            'bike-name' => 'required',
            'bike-brand' => 'required',
            'bike-price' => ['required', 'integer'],
        ]);
        $bike = new Bike();
        $bike->name = strip_tags($request->input('bike-name'));
        $bike->brand = strip_tags($request->input('bike-brand'));
        $bike->price = strip_tags($request->input('bike-price'));
        $bike->save();
        return redirect()->route('bikes.index');
    }
```
	Request객체의 validate()메소드로 값의 필수값을 설정할 수 있다.
	Request객체로 전달받은 form태그의 값을 모델 객체에 넣어줄 때 strip_tags()메소드를 이용
	해 사이트 스크립트 공격을 방지할 수 있다.

```php
	public function update(BikeFormRequest $request, Bike $bike)
	//파라미터 Request자리에 사용자 정의 Request객체 사용
	{
		//Request의 validated()메소드로 값 검증
		$data = $request->validated();
		$bike->name = $data['bike-name'];
		$bike->brand = $data['bike-brand'];
		$bike->price = $data['bike-price'];
		$bike->save();
		return redirect()->route('bikes.show', $bike->id);
	
	}
```
