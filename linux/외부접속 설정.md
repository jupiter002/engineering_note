sudo systemctl enable ssh

//방화벽 설정
sudo ufw status //방화벽 상태 확인
sudo ufw enable //방화벽 활성화
sudo ufw disable //방화벽 비활성화

//방화벽 내 ssh설정
//허용
sudo ufw allow ssh // ssh방화벽 허용
sudo ufw allow 22/tcp // tcp만 허용
sudo ufw allow 22/udp // udp만 허용

//거부
sudo ufw deny  22 // tcp/udp 22번 모두 거부
sudo ufw deny 22/tcp // 22번 포트 tcp만 거부
sudo ufw deny 22/udp // 22번 포트 udp만 거부
